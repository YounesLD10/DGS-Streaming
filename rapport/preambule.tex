% --- Numerotation des chapitres : "Chapitre N" -----------------------------

% --- Titre 1 (chapitre) : fond vert, texte blanc, Arial 18pt gras ----------
\titleformat{\chapter}[block]
  {\headfont\fontsize{18}{22}\selectfont\bfseries\color{white}}
  {}
  {0pt}
  {%
    \begin{tikzpicture}[remember picture, overlay]
      \node[fill=EMSIvert, anchor=north west, minimum width=\textwidth,
            minimum height=1.6cm, text width=\textwidth-0.6cm,
            align=left, inner sep=0.3cm]
        at ($(current page.north west)+(2.54cm,-3.0cm)$)
        {\headfont\fontsize{18}{22}\selectfont\bfseries\color{white}%
         Chapitre \thechapter\ ---\ #1};
    \end{tikzpicture}%
    \vspace{1.4cm}%
  }
\titlespacing*{\chapter}{0pt}{0pt}{30pt}
